cmake_minimum_required(VERSION 3.21)

project(AppGradientTerrains LANGUAGES CXX)

# ----------------------------------------------------------------------------------------------------------------------
# Gradient terrain editor and the four libraries it uses (Libs/).
#
#   Windows / Visual Studio :  cmake -S . -B build -G "Visual Studio 17 2022" -A x64 -DCMAKE_PREFIX_PATH=<Qt6 msvc dir>
#                              cmake --build build --config Release
#   Linux / Makefiles       :  cmake -S . -B build -DCMAKE_BUILD_TYPE=Release
#                              cmake --build build -j
#
# Needs Qt 6 (Core, Gui, Widgets, OpenGLWidgets), OpenGL with GLU and GLEW. OpenGL 4.3 (compute shaders) is
# required at run time. The Visual Studio solution and the .vcxproj files of the repository are not used here.
# ----------------------------------------------------------------------------------------------------------------------

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

# One folder for every binary: the executable finds its shaders and icons next to it
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

set(CMAKE_AUTOMOC ON)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)

# ---- Qt ---------------------------------------------------------------------------------------------------------------
# QTDIR64 is the variable already used by the Visual Studio projects of this repository
if(DEFINED ENV{QTDIR64})
  list(APPEND CMAKE_PREFIX_PATH "$ENV{QTDIR64}")
endif()
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Widgets OpenGLWidgets)

# ---- OpenGL, GLU, GLEW, OpenMP ----------------------------------------------------------------------------------------
find_package(OpenGL REQUIRED)
find_package(OpenMP)

if(WIN32)
  # GLEW shipped with the libraries
  set(GLEW_DIR_LOCAL ${CMAKE_SOURCE_DIR}/Libs/LibPlatform/LibGL/x64)
  add_library(GLEW::GLEW SHARED IMPORTED)
  set_target_properties(GLEW::GLEW PROPERTIES
    IMPORTED_IMPLIB ${GLEW_DIR_LOCAL}/Library/glew32.lib
    IMPORTED_LOCATION ${GLEW_DIR_LOCAL}/bin/glew32.dll
    INTERFACE_INCLUDE_DIRECTORIES ${GLEW_DIR_LOCAL}/Include)
  set(GLU_LIBRARY glu32)
else()
  find_package(GLEW REQUIRED)
  if(TARGET OpenGL::GLU)
    set(GLU_LIBRARY OpenGL::GLU)
  else()
    set(GLU_LIBRARY GLU)
  endif()
endif()

# ---- Options common to every target ----------------------------------------------------------------------------------
function(arches_target_options target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /Zc:__cplusplus /permissive- /MP
      $<$<CONFIG:Release>:/arch:AVX> $<$<CONFIG:Release>:/fp:fast>)
  else()
    target_compile_options(${target} PRIVATE -mavx $<$<CONFIG:Release>:-ffast-math>)
  endif()
  if(OpenMP_CXX_FOUND)
    target_link_libraries(${target} PUBLIC OpenMP::OpenMP_CXX)
  endif()
endfunction()

# The Visual Studio projects force-include a header in every source file (precompiled header). GCC does not honour
# '#pragma once' for headers that are part of a precompiled header, so there it is a plain forced include.
function(arches_force_include target header)
  if(MSVC)
    target_precompile_headers(${target} PRIVATE ${header})
  else()
    target_compile_options(${target} PRIVATE -include ${header})
  endif()
endfunction()

# ---- Libraries (Libs/<Name>/<Name>/{Include,Source}) ------------------------------------------------------------------
# arches_add_library(<Name> PCH <forced include header> LINK <targets...>)
# The .cpp files whose name ends with -qt.cpp only create the precompiled header of the Visual Studio projects, and
# *-moc.cpp are moc files generated by those projects: both are replaced by CMake's own precompiled header and AUTOMOC.
function(arches_add_library name)
  cmake_parse_arguments(A "" "PCH" "LINK" ${ARGN})
  set(dir ${CMAKE_SOURCE_DIR}/Libs/${name}/${name})
  file(GLOB sources CONFIGURE_DEPENDS ${dir}/Source/*.cpp)
  list(FILTER sources EXCLUDE REGEX "-qt\\.cpp$")
  list(FILTER sources EXCLUDE REGEX "-moc\\.cpp$")
  file(GLOB headers CONFIGURE_DEPENDS ${dir}/Include/*.h)
  add_library(${name} STATIC ${sources} ${headers})
  target_include_directories(${name} PUBLIC ${dir}/Include)
  target_link_libraries(${name} PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets Qt6::OpenGLWidgets
                                       GLEW::GLEW OpenGL::GL ${GLU_LIBRARY} ${A_LINK})
  if(A_PCH)
    arches_force_include(${name} ${dir}/Include/${A_PCH})
  endif()
  arches_target_options(${name})
  set_target_properties(${name} PROPERTIES FOLDER "Libs")
endfunction()

arches_add_library(LibCore           PCH core-qt.h)
arches_add_library(LibHeightfield    PCH heightfield-qt.h          LINK LibCore)
arches_add_library(LibHeightAnalysis PCH heightfield-analysis-qt.h LINK LibHeightfield LibCore)
arches_add_library(LibMaya           PCH maya-qt.h                 LINK LibHeightfield LibCore)

# ---- Application ------------------------------------------------------------------------------------------------------
set(APP_DIR ${CMAKE_SOURCE_DIR}/AppGradientTerrains)
file(GLOB APP_SOURCES CONFIGURE_DEPENDS ${APP_DIR}/Source/*.cpp)
list(FILTER APP_SOURCES EXCLUDE REGEX "moc_[^/\\\\]*\\.cpp$")   # generated by AUTOMOC
list(FILTER APP_SOURCES EXCLUDE REGEX "app-qt\\.cpp$")          # precompiled header of the Visual Studio project
set(APP_HEADERS
  ${APP_DIR}/Include/qte.h ${APP_DIR}/Include/gradientterrain.h
  ${APP_DIR}/Include/diffusionterrain.h ${APP_DIR}/Include/tools-render.h)

# The header generated from the Qt Designer file keeps the name it has in the Visual Studio project (uic_<name>.h)
set(UI_FILE ${APP_DIR}/UI/mainWindowProjetQT.ui)
set(UI_HEADER ${CMAKE_BINARY_DIR}/generated/uic_mainWindowProjetQT.h)
add_custom_command(
  OUTPUT ${UI_HEADER}
  COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_BINARY_DIR}/generated
  COMMAND Qt6::uic ${UI_FILE} -o ${UI_HEADER}
  DEPENDS ${UI_FILE} Qt6::uic
  COMMENT "Compiling mainWindowProjetQT.ui"
  VERBATIM)

add_executable(AppGradient ${APP_SOURCES} ${APP_HEADERS} ${UI_HEADER} ${UI_FILE})
# generated/ comes first so that the freshly generated header hides a stale copy of the Visual Studio build
target_include_directories(AppGradient PRIVATE ${CMAKE_BINARY_DIR}/generated ${APP_DIR}/Include)
target_link_libraries(AppGradient PRIVATE LibMaya LibHeightAnalysis LibHeightfield LibCore
                                          Qt6::Core Qt6::Gui Qt6::Widgets Qt6::OpenGLWidgets
                                          OpenGL::GL ${GLU_LIBRARY})
# main.cpp falls back on this folder when it cannot find the Libs folder above the executable
target_compile_definitions(AppGradient PRIVATE ARCHES_LIBS_DIR="${CMAKE_SOURCE_DIR}/Libs")
arches_force_include(AppGradient ${APP_DIR}/Include/qte.h)
arches_target_options(AppGradient)

# Shaders and icons are read relative to the working directory: copy them next to the executable
add_custom_command(TARGET AppGradient POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${APP_DIR}/Shaders $<TARGET_FILE_DIR:AppGradient>/Shaders
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${APP_DIR}/UI/Icons $<TARGET_FILE_DIR:AppGradient>/UI/Icons
  COMMAND ${CMAKE_COMMAND} -E copy_if_different ${APP_DIR}/1025.png $<TARGET_FILE_DIR:AppGradient>
  COMMENT "Copying shaders, icons and the 1025x1025 template")

if(MSVC)
  set_property(TARGET AppGradient PROPERTY VS_DEBUGGER_WORKING_DIRECTORY $<TARGET_FILE_DIR:AppGradient>)
  set_property(DIRECTORY ${CMAKE_SOURCE_DIR} PROPERTY VS_STARTUP_PROJECT AppGradient)
endif()

# ---- Qt and GLEW DLLs next to the executable (Windows) ----------------------------------------------------------------
if(WIN32)
  get_target_property(_qmake Qt6::qmake IMPORTED_LOCATION)
  get_filename_component(_qt_bin "${_qmake}" DIRECTORY)
  find_program(WINDEPLOYQT windeployqt HINTS "${_qt_bin}")
  add_custom_command(TARGET AppGradient POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${GLEW_DIR_LOCAL}/bin/glew32.dll $<TARGET_FILE_DIR:AppGradient>
    COMMENT "Copying glew32.dll")
  if(WINDEPLOYQT)
    add_custom_command(TARGET AppGradient POST_BUILD
      # Minimal deployment: the application only reads PNG images and uses the built-in Fusion style
      COMMAND ${WINDEPLOYQT} --no-translations --no-system-d3d-compiler --no-system-dxc-compiler --no-opengl-sw
              --no-compiler-runtime
              --skip-plugin-types generic,iconengines,imageformats,networkinformation,styles,tls
              $<IF:$<CONFIG:Debug>,--debug,--release> $<TARGET_FILE:AppGradient>
      COMMENT "Deploying the Qt libraries")
  endif()
endif()
